def _breadcrumb(items):
  parts = ""
  for i,(label,href) in enumerate(items):
    if i>0: parts += '<li class="rsb-breadcrumb__sep">/</li>'
    if href is None: parts += '<li><span class="rsb-breadcrumb__current">%s</span></li>'%label
